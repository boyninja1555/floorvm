#define REG_DECREMENTAL r1
#define REG_TIMER r0

_start:
	set32 ${REG_DECREMENTAL} 1  ; Sets timer decremental to 1
	set32 ${REG_TIMER} 500000   ; Sets initial timer to 500000

loop:
	sub ${REG_TIMER} ${REG_TIMER} ${REG_DECREMENTAL}  ; Subtracts 1 from timer
	jmpnz loop ${REG_TIMER}                           ; Restarts loop if timer != 0

halt  ; Automatically halts execution when loop is escaped
