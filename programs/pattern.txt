#define ADDR_COLOR 3252
#define ADDR_COUNTDOWN 3248
#define ADDR_VRAM 2048
#define VRAM_SIZE 1200
#define REG_POS r0
#define REG_TEMP r1
#define REG_TMP2 r2

set32 ${REG_POS} ${ADDR_VRAM}

set32 ${REG_TEMP} 1200                 ; Sets r1 to 1200
store32 ${REG_TEMP} ${ADDR_COUNTDOWN}  ; Stores 1200 (r1) into countdown

pattern:
	load8 ${ADDR_COLOR} ${REG_TEMP}
	set32 ${REG_TMP2} 10
	add ${REG_TMP2} ${REG_TEMP} ${REG_TMP2}
	store8 ${REG_TMP2} ${ADDR_COLOR}

	add ${REG_TEMP} ${REG_TEMP} ${REG_TEMP}  ; Sets r1 to 255
	store8p ${REG_TEMP} ${REG_POS}           ; Stores 255 (r1) into position (r0)

	set32 ${REG_TEMP} 1                      ; Sets r1 to 1
	add ${REG_POS} ${REG_POS} ${REG_TEMP}    ; Adds 1 (r1) to position (r0)

	load32 ${ADDR_COUNTDOWN} ${REG_TEMP}     ; Loads countdown into r1
	set32 ${REG_TMP2} 1                      ; Sets r2 to 1
	sub ${REG_TEMP} ${REG_TEMP} ${REG_TMP2}  ; Subtracts 1 (r2) from countdown
	store32 ${REG_TEMP} ${ADDR_COUNTDOWN}    ; Stores r1 into countdown
	jmpnz pattern ${REG_TEMP}                ; Next pattern run if countdown != zero

forever:         ; Automatically jumped to after pattern is drawn (due to label ordering)
	jmp forever  ; Restarts forever loop
